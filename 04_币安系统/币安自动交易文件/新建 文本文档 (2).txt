def get_data_cached(sym, tf):
    now = time.time(); cache_key = f"{sym}_{tf}"
    cache_time = 120 if tf == '15m' else (300 if tf == '1h' else 600)
    if cache_key in DATA_CACHE and (now - DATA_CACHE[cache_key]['time']) < cache_time: return DATA_CACHE[cache_key]['data']
    try:
        ohlcv = exchange.fetch_ohlcv(sym, tf, limit=500)
        df = pd.DataFrame(ohlcv, columns=['t','o','h','l','c','v'])
        if tf == '4h': df['ema576'] = df['c'].ewm(span=576).mean()
        elif tf == '1h':
            df['ema12'], df['ema26'] = df['c'].ewm(span=12, adjust=False).mean(), df['c'].ewm(span=26, adjust=False).mean()
            df['macd_hist'] = (df['ema12'] - df['ema26']) - (df['ema12'] - df['ema26']).ewm(span=9, adjust=False).mean()
        elif tf == '15m':
            df['ema14'], df['ema21'], df['ema60'] = df['c'].ewm(span=14).mean(), df['c'].ewm(span=21).mean(), df['c'].ewm(span=60).mean()
            df['vol_sma20'] = df['v'].rolling(20).mean(); df['prev_c'] = df['c'].shift(1)
            df['tr'] = df[['h','l','prev_c']].apply(lambda x: max(x['h']-x['l'], abs(x['h']-x['prev_c']), abs(x['l']-x['prev_c'])) if not pd.isna(x['prev_c']) else x['h']-x['l'], axis=1)
            df['atr'] = df['tr'].ewm(alpha=1/14).mean()
        DATA_CACHE[cache_key] = {'time': now, 'data': df}; return df
    except: return None

def assassin_fire(sym, side, nom_u, lev):
    try:

        p = exchange.fetch_ticker(sym).get('last')

        exchange.create_market_order(sym, 'buy' if side=='buy' else 'sell', amt)
        return True
    except Exception as e: logger.error(f"❌ 进攻受阻: {e}"); return False

def armor_close(sym, side, amt):

        exchange.create_market_order(sym, 'sell' if side == 'long' else 'buy', amt, params={'reduceOnly': True})
