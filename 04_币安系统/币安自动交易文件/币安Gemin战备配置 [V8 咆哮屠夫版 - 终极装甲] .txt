# -*- coding: utf-8 -*-
import os
import time
import pandas as pd
import ccxt
from dotenv import load_dotenv
from datetime import timedelta

# ==================== 1. 战备配置 [V8 咆哮屠夫版 - 终极装甲] ====================
load_dotenv()

MAJORS_BASE = ['BTC', 'ETH', 'SOL', 'BNB', 'NEAR', 'AVAX']
ALTS_BASE = ['DOGE', 'WLD', 'HYPE', 'LTC', 'BCH', 'ARB']
SYMBOL_LIST = [f"{s}/USDT" for s in (MAJORS_BASE + ALTS_BASE)]

INITIAL_HARD_SL = 0.030    
SOFT_STOP_OFFSET = 0.002   
EQUITY_PROTECT_LINE = 0.10 
EQUITY_PROTECT_ADD = 0.002 
MARGIN_SAFE_RATIO = 0.30   
MAX_TOTAL_MARGIN = 0.35    
MAX_POSITIONS = 7          

LEVEL_LOG = {}             
HIGH_WATER_MARK = {}       
POSITION_TIMERS = {}      
ACTUAL_LEVERAGE = {}       
SYNCED_POSITIONS = set()   
INITIAL_WALLET = None      
PEAK_WALLET = 0            
SYSTEM_START_TIME = time.time()  
LAST_DISPLAY_TIME = 0     
LAST_SCAN_TIME = 0         

# ==================== 2. 交易所初始化 ====================
exchange = ccxt.binance({
    'apiKey': os.getenv('BINANCE_API_KEY'),
    'secret': os.getenv('BINANCE_SECRET'),
    'options': {'defaultType': 'future'},
    'enableRateLimit': True,
    'adjustForTimeDifference': True,
    'proxies': {
        'http': 'http://127.0.0.1:7897', 
        'https': 'http://127.0.0.1:7897'
    }
})

exchange.enable_demo_trading(True) 

print(f"[SYSTEM] V8 咆哮屠夫版(终极装甲) 启动！网络信道[7897]接驳中...")

try:
    exchange.set_position_mode(hedged=False)
except Exception:
    pass

# ==================== 3. 核心全息探针 ====================
def check_is_major(symbol):
    base = symbol.split('/')[0].split(':')[0].upper()
    return base in MAJORS_BASE

def normalize_symbol(s):
    return s.replace('/', '').split(':')[0].replace('USDT', '')

def get_real_leverage(pos):
    lev = pos.get('leverage')
    if lev is not None:
        return int(lev)
    info_lev = pos.get('info', {}).get('leverage')
    if info_lev is not None:
        return int(info_lev)
    return 1

def get_real_pnl(pos):
    pnl = pos.get('unrealizedPnl')
    if pnl is not None:
        return float(pnl)
    info_pnl = pos.get('info', {}).get('unRealizedProfit')
    if info_pnl is not None:
        return float(info_pnl)
    return 0.0

def get_processed_data(symbol, timeframe):
    try:
        ohlcv = exchange.fetch_ohlcv(symbol, timeframe=timeframe, limit=150)
        df = pd.DataFrame(ohlcv, columns=['t', 'o', 'h', 'l', 'c', 'v'])
        df['ema9'] = df['c'].ewm(span=9, adjust=False).mean()
        df['ema21'] = df['c'].ewm(span=21, adjust=False).mean()
        if timeframe == '4h':
            df['ema144'] = df['c'].ewm(span=144, adjust=False).mean()
            df['ema169'] = df['c'].ewm(span=169, adjust=False).mean()
        return df
    except Exception:
        return None

def butcher_fire(symbol, side, amount_u, leverage, current_lvl):
    try:
        if current_lvl == 0:
            try: 
                exchange.set_leverage(leverage, symbol)
                ACTUAL_LEVERAGE[normalize_symbol(symbol)] = leverage
                time.sleep(0.2)
            except Exception:
                pass
            
        ticker = exchange.fetch_ticker(symbol)
        price = ticker['last']
        amount = float(exchange.amount_to_precision(symbol, (amount_u * leverage) / price))
        
        exchange.create_market_order(symbol, side, amount)
        print(f"🚀 [FIRE] L{current_lvl}->L{current_lvl+1} | {symbol} | {leverage}x | {amount_u:.1f}U")
        return True
    except Exception as e:
        print(f"❌ [FAILED] 开火失败 {symbol}: {e}")
        return False

# ==================== 4. 战术主循环 ====================
def run_butcher_room():
    global INITIAL_WALLET, PEAK_WALLET, LAST_DISPLAY_TIME, LAST_SCAN_TIME
    
    try:
        current_time = time.time()
        bal = exchange.fetch_balance(params={'type': 'future'})
        wallet = float(bal['total'].get('USDT', 0))
        available = float(bal['free'].get('USDT', 0))
        pnl = float(bal['info'].get('totalUnrealizedProfit', 0))

        if INITIAL_WALLET is None: 
            INITIAL_WALLET = wallet
        if wallet > PEAK_WALLET: 
            PEAK_WALLET = wallet

        available_ratio = available / wallet if wallet > 0 else 0
        margin_used_ratio = (wallet - available) / wallet if wallet > 0 else 0
        CAN_ADD = True if (available_ratio > MARGIN_SAFE_RATIO and margin_used_ratio < MAX_TOTAL_MARGIN) else False

        profit_ratio = (wallet - INITIAL_WALLET) / INITIAL_WALLET if INITIAL_WALLET > 0 else 0
        protect_shift = EQUITY_PROTECT_ADD if profit_ratio >= EQUITY_PROTECT_LINE else 0

        all_pos = exchange.fetch_positions()
        current_positions = {normalize_symbol(p['symbol']): p for p in all_pos if abs(float(p.get('contracts', 0))) > 0}

        # --- 1. 存量头寸同步与级别识别 (严格区分兵力基准) ---
        for norm, pos in current_positions.items():
            full_sym = pos['symbol']
            is_major = check_is_major(full_sym)
            target_lev = 20 if is_major else 5

            if norm not in SYNCED_POSITIONS:
                try: 
                    exchange.set_leverage(target_lev, full_sym)
                    ACTUAL_LEVERAGE[norm] = target_lev
                    SYNCED_POSITIONS.add(norm)
                except Exception:
                    pass

            if norm not in LEVEL_LOG:
                lev = ACTUAL_LEVERAGE.get(norm, get_real_leverage(pos))
                amt = abs(float(pos.get('contracts', 0)))
                entry = float(pos.get('entryPrice', 0))
                notional = entry * amt
                margin = notional / lev if lev > 0 else notional
                
                # 🎯 终极打磨：严格区分大小币种的基数，根除重启误判
                base_ratio = 0.015 if is_major else 0.010
                base_u = INITIAL_WALLET * base_ratio if INITIAL_WALLET else (75 if is_major else 50)
                
                if margin > (base_u * 3):
                    LEVEL_LOG[norm] = 3
                elif margin > (base_u * 1.5):
                    LEVEL_LOG[norm] = 2
                else:
                    LEVEL_LOG[norm] = 1
                    
                POSITION_TIMERS[norm] = current_time

        # --- 2. 战术大屏 ---
        if current_time - LAST_DISPLAY_TIME >= 45:
            print("\n" + "="*125)
            run_dur = str(timedelta(seconds=int(current_time-SYSTEM_START_TIME)))[:5]
            print(f" [V8 咆哮屠夫] | 阵地:{len(current_positions)}/{MAX_POSITIONS} | 防御金:{available_ratio:.1%} | 运行: {run_dur}")
            print(f" [资产] 初始:{INITIAL_WALLET:.2f}U | 当前:{wallet:.2f}U | 浮盈:{pnl:+.2f}U | 净利比:{profit_ratio:.2%}")
            print("="*125)
            print(" 战区(币种)   | 方向 | 级别 | 杠杆 | 占用本金(U)  | 仓位价值(U)  | 收益率     | 潜伏时长   | 状态")
            print("-" * 125)
            for norm, pos in current_positions.items():
                side_str = "多" if pos.get('side') == 'long' else "空"
                lev = ACTUAL_LEVERAGE.get(norm, get_real_leverage(pos))
                
                amt = abs(float(pos.get('contracts', 0)))
                entry = float(pos.get('entryPrice', 0))
                notional = entry * amt
                margin = notional / lev if lev > 0 else notional
                pnl_u = get_real_pnl(pos)
                roi = (pnl_u / margin * 100) if margin > 0 else 0
                dur = str(timedelta(seconds=int(current_time - POSITION_TIMERS.get(norm, current_time))))[0:5]
                lvl = LEVEL_LOG.get(norm, 1)
                print(f" {norm:<12} | {side_str:<2}   | {lvl:<2}   | {lev:<3}x | {margin:<12.2f} | {notional:<12.2f} | {roi:>+8.2f}%   | {dur:<10} | L{lvl}重炮锁定")
            LAST_DISPLAY_TIME = current_time

        # --- 3. 存量风控与加仓管理 (带独立防爆装甲) ---
        for norm, pos in list(current_positions.items()):
            full_sym = pos['symbol']
            side = pos.get('side', 'long').lower()
            amt = abs(float(pos.get('contracts', 0)))
            entry = float(pos.get('entryPrice', 0))
            mark = float(pos.get('markPrice', 0))
            
            is_major = check_is_major(full_sym)
            p_move = (mark - entry) / entry if side == 'long' else (entry - mark) / entry
            lvl = LEVEL_LOG.get(norm, 1)
            lev = ACTUAL_LEVERAGE.get(norm, get_real_leverage(pos))

            order_side = 'buy' if side == 'long' else 'sell'
            close_side = 'sell' if side == 'long' else 'buy'

            # --- 加仓模块 ---
            if CAN_ADD:
                if is_major:
                    if lvl == 1 and p_move >= 0.015:
                        if butcher_fire(full_sym, order_side, wallet * 0.025, lev, 1): 
                            LEVEL_LOG[norm] = 2
                    elif lvl == 2 and p_move >= 0.030:
                        if butcher_fire(full_sym, order_side, wallet * 0.045, lev, 2): 
                            LEVEL_LOG[norm] = 3
                else: 
                    if lvl == 1 and p_move >= 0.025:
                        if butcher_fire(full_sym, order_side, wallet * 0.01, lev, 1): 
                            LEVEL_LOG[norm] = 2
                    elif lvl == 2 and p_move >= 0.050:
                        if butcher_fire(full_sym, order_side, wallet * 0.015, lev, 2): 
                            LEVEL_LOG[norm] = 3

            # --- 止损/硬防御模块 (独立装甲保护) ---
            current_sl = -INITIAL_HARD_SL if lvl == 1 else SOFT_STOP_OFFSET
            final_sl = current_sl + protect_shift
            
            if p_move < final_sl:
                try:
                    exchange.create_market_order(full_sym, close_side, amt, params={'reduceOnly': True})
                    print(f"🗡️ [KILL] {norm} 触及防线离场。SL: {final_sl:.2%}")
                    LEVEL_LOG.pop(norm, None)
                    HIGH_WATER_MARK.pop(norm, None)
                    POSITION_TIMERS.pop(norm, None)
                    if norm in SYNCED_POSITIONS: 
                        SYNCED_POSITIONS.remove(norm)
                except Exception as e:
                    print(f"⚠️ [WARNING] {norm} 离场受阻，防线网络卡顿，等待重试: {e}")
                continue

            # --- L3 绞肉机收网模块 (独立装甲保护) ---
            if lvl == 3:
                if norm not in HIGH_WATER_MARK: 
                    HIGH_WATER_MARK[norm] = mark
                    
                if side == 'long': 
                    HIGH_WATER_MARK[norm] = max(HIGH_WATER_MARK[norm], mark)
                else: 
                    HIGH_WATER_MARK[norm] = min(HIGH_WATER_MARK[norm], mark)
                
                limit = 0.004 if is_major else 0.008
                drawdown = (HIGH_WATER_MARK[norm] - mark)/HIGH_WATER_MARK[norm] if side == 'long' else (mark - HIGH_WATER_MARK[norm])/HIGH_WATER_MARK[norm]
                
                if drawdown >= limit:
                    try:
                        exchange.create_market_order(full_sym, close_side, amt, params={'reduceOnly': True})
                        print(f"💰 [PROFIT] {norm} 追踪收网成功！利润落袋。")
                        LEVEL_LOG.pop(norm, None)
                        HIGH_WATER_MARK.pop(norm, None)
                        POSITION_TIMERS.pop(norm, None)
                        if norm in SYNCED_POSITIONS: 
                            SYNCED_POSITIONS.remove(norm)
                    except Exception as e:
                        print(f"⚠️ [WARNING] {norm} 收网指令超时，利润正在撤离: {e}")

        # --- 4. 雷达扫描 ---
        if current_time - LAST_SCAN_TIME >= 60 and len(current_positions) < MAX_POSITIONS and CAN_ADD:
            LAST_SCAN_TIME = current_time
            active_count = len(current_positions)
            
            for sym in SYMBOL_LIST:
                norm = normalize_symbol(sym)
                if norm in current_positions: 
                    continue
                
                if active_count >= MAX_POSITIONS: 
                    break
                
                df4h = get_processed_data(sym, '4h')
                df15 = get_processed_data(sym, '15m')
                if df4h is None or df15 is None: 
                    continue
                
                vh = max(df4h['ema144'].iloc[-1], df4h['ema169'].iloc[-1])
                vl = min(df4h['ema144'].iloc[-1], df4h['ema169'].iloc[-1])
                cp = df15['c'].iloc[-1]
                e9 = df15['ema9'].iloc[-1]
                e21 = df15['ema21'].iloc[-1]
                rh = df15['h'].iloc[-11:-1].max()
                rl = df15['l'].iloc[-11:-1].min()
                
                is_major = check_is_major(sym)
                ratio, lev = (0.015, 20) if is_major else (0.01, 5)
                
                if cp > vh and cp > rh and e9 > e21:
                    if butcher_fire(sym, 'buy', wallet * ratio, lev, 0):
                        LEVEL_LOG[norm] = 1
                        POSITION_TIMERS[norm] = current_time
                        ACTUAL_LEVERAGE[norm] = lev
                        SYNCED_POSITIONS.add(norm)
                        active_count += 1
                elif cp < vl and cp < rl and e9 < e21:
                    if butcher_fire(sym, 'sell', wallet * ratio, lev, 0):
                        LEVEL_LOG[norm] = 1
                        POSITION_TIMERS[norm] = current_time
                        ACTUAL_LEVERAGE[norm] = lev
                        SYNCED_POSITIONS.add(norm)
                        active_count += 1
                time.sleep(0.3)
                
    except Exception as e:
        print(f"🚨 [CRITICAL] 核心循环受到强烈干扰，防爆机制已启动: {e}")

# ==================== 5. 主循环 ====================
if __name__ == "__main__":
    try:
        while True:
            run_butcher_room()
            time.sleep(10)
    except KeyboardInterrupt:
        print("\n🛑 [SYSTEM] 接收到统帅撤退指令，屠夫系统已安全挂起并退出！")
