def load_or_train_model():
    global GLOBAL_AI_MODEL
    if os.path.exists(MODEL_FILE):
        GLOBAL_AI_MODEL = joblib.load(MODEL_FILE)
        print("✅ [AI] RandomForest 模型已加载")
        return
    print("🧠 [AI] 正在拉取真实战场数据铸造 AI 智脑...")
    try:
        ohlcv = exchange.fetch_ohlcv('BTC/USDT', '15m', limit=1000)
        df = pd.DataFrame(ohlcv, columns=['t','o','h','l','c','v'])
        df['vol_sma'] = df['v'].rolling(20).mean()
        df['ema144'] = df['c'].ewm(span=144).mean()
        df['bias'] = abs(df['c'] - df['ema144']) / (df['ema144'] + 1e-9)
        df['mom'] = df['c'].pct_change(5)
        df['vol_r'] = df['v'] / (df['vol_sma'] + 1e-9)
        df['target'] = (df['c'].shift(-5) - df['c']) / df['c'] * 1000 + 70
        df.dropna(inplace=True)
        X = df[['bias', 'vol_r', 'mom']]
        y = df['target'].clip(0, 100)
        GLOBAL_AI_MODEL = RandomForestRegressor(n_estimators=30, max_depth=5, random_state=42).fit(X, y)
        joblib.dump(GLOBAL_AI_MODEL, MODEL_FILE)
        print("✅ [AI] 真实数据模型训练完成")
    except Exception as e:
        print(f"⚠️ 真实数据训练失败 ({e})，启用备用...")
        X = pd.DataFrame(np.random.rand(200, 3), columns=['bias', 'vol_r', 'mom'])
        y = 70 + (X['vol_r'] * 15) - (X['bias'] * 30) + (X['mom'] * 10)
        GLOBAL_AI_MODEL = RandomForestRegressor(n_estimators=20, max_depth=4).fit(X, y)
        joblib.dump(GLOBAL_AI_MODEL, MODEL_FILE)
    if GLOBAL_AI_MODEL is None:
        GLOBAL_AI_MODEL = RandomForestRegressor(n_estimators=20, max_depth=4).fit(
            pd.DataFrame(np.random.rand(200, 3), columns=['bias', 'vol_r', 'mom']),
            np.full(200, 70)
        )

def update_funding_rates():
    global GLOBAL_FUNDING_RATES, LAST_FUNDING_FETCH
    if time.time() - LAST_FUNDING_FETCH > 3600:
        try:
            frs = exchange.fetch_funding_rates()
            for s, data in frs.items():
                GLOBAL_FUNDING_RATES[s] = abs(float(data.get('fundingRate', 0)))
            LAST_FUNDING_FETCH = time.time()
        except: pass

def get_funding_rate_cached(sym):
    return GLOBAL_FUNDING_RATES.get(sym, 0.0)

def get_ls_ratio(sym):
    try:
        res = exchange.fapiPublicGetGlobalLongShortAccountRatio({'symbol': sym.replace('/',''), 'period': '15m', 'limit': 1})
        return float(res[0]['longShortRatio'])
    except: return 1.0

def get_data_cached(sym, tf):
    now = time.time()
    cache_key = f"{sym}_{tf}"
    ttl = 180 if tf == '15m' else 900
    if cache_key in DATA_CACHE and (now - DATA_CACHE[cache_key]['time']) < ttl:
        return DATA_CACHE[cache_key]['data']
    try:
        ohlcv = exchange.fetch_ohlcv(sym, tf, limit=1000 if tf == '4h' else 500)
        df = pd.DataFrame(ohlcv, columns=['t','o','h','l','c','v'])
        if tf == '4h':
            df['ema144'] = df['c'].ewm(span=144).mean()
            df['ema169'] = df['c'].ewm(span=169).mean()
            df['ema576'] = df['c'].ewm(span=576).mean()
        if tf == '15m':
            df['ema14'] = df['c'].ewm(span=14).mean()
            df['ema28'] = df['c'].ewm(span=28).mean()
            df['ema20_p'] = df['c'].ewm(span=20).mean()
            df['ema144'] = df['c'].ewm(span=144).mean()
            df['vol_sma20'] = df['v'].rolling(20).mean()
            df['h10'] = df['h'].rolling(10).max().shift(1)
            df['l10'] = df['l'].rolling(10).min().shift(1)
            df['tr'] = df[['h','l','c']].apply(lambda x: max(x['h']-x['l'], abs(x['h']-x['c']), abs(x['l']-x['c'])), axis=1)
            df['atr'] = df['tr'].ewm(alpha=1/14).mean()
        DATA_CACHE[cache_key] = {'time': now, 'data': df}
        return df
    except:
        return None

def clear_state(n, is_cooldown=True):
    for d in [LEVEL_LOG, BASE_SIZE_LOG, HIGH_WATER_MARK, POSITION_TIMERS, ENTRY_PRICE_LOG, ACTUAL_LEVERAGE, DYNAMIC_SL_LOG, PARTIAL_CLOSED, ENTRY_ATR_LOG]:
        d.pop(n, None)
    SYNCED_POSITIONS.discard(n)
    if is_cooldown:
        COOLDOWN_LOG[n] = time.time() + COOLDOWN_TIME

def assassin_fire(sym, side, nom_u, lev):
    n = sym.replace('/USDT','')
    if n in LEVEL_LOG: return False
    try:
        try: exchange.set_leverage(lev, sym)
        except: pass
        p = exchange.fetch_ticker(sym).get('last')
        amt = float(exchange.amount_to_precision(sym, nom_u / p))
        exchange.create_market_order(sym, 'buy' if side=='buy' else 'sell', amt)
        print(f"🔫 [ASSASSIN FIRE] 果断开火! {sym} | {side.upper()} | {lev}x")
        return True
    except: return False

def armor_close(sym, side, amt):
    action = 'sell' if side == 'long' else 'buy'
    try:
        exchange.create_market_order(sym, action, amt, params={'reduceOnly': True})
        return True
    except: return False

def run_assassin_room():
    global INITIAL_WALLET, DAILY_START_WALLET, DAILY_DATE, LAST_DISPLAY, SCAN_INDEX, CAN_TRADE_TODAY
    try:
        update_funding_rates()

        cur_t = time.time()
        cur_d = datetime.now().strftime('%Y-%m-%d')
        bal = exchange.fetch_balance(params={'type':'future'})
        raw_pos = exchange.fetch_positions()
        wallet = float(bal.get('total',{}).get('USDT') or 0)
        available = float(bal.get('free',{}).get('USDT') or 0)
        cur_pos = {p['symbol'].replace('/USDT',''): p for p in raw_pos if abs(float(p.get('contracts') or 0)) > 0}
        total_pnl = sum([float(p.get('unrealizedPnl', 0)) if 'unrealizedPnl' in p else float(p.get('info', {}).get('unRealizedProfit', 0)) for p in cur_pos.values()])
        equity = wallet + total_pnl

        if INITIAL_WALLET is None: INITIAL_WALLET = equity
        if DAILY_START_WALLET is None or DAILY_DATE != cur_d:
            DAILY_DATE = cur_d
            DAILY_START_WALLET = equity
            CAN_TRADE_TODAY = True

        daily_pnl = (equity - DAILY_START_WALLET) / (DAILY_START_WALLET + 1e-9)

        if daily_pnl <= -DAILY_MAX_LOSS:
            CAN_TRADE_TODAY = False
            if len(cur_pos) > 0:
                for n, p in cur_pos.items():
                    armor_close(p['symbol'], p['side'].lower(), abs(float(p.get('contracts',0))))
                    clear_state(n, is_cooldown=True)
                save_state()
            return

        if cur_t - LAST_DISPLAY >= 10:
            console.clear()
            console.print("[dim]" + "-"*115 + "[/dim]")
            console.print(f" [bold cyan][帝国版图][/bold cyan] 初始:{INITIAL_WALLET:.2f}U | 当日起点:{DAILY_START_WALLET:.2f}U | 净利:{(equity-INITIAL_WALLET)/INITIAL_WALLET:+.2%}")
            console.print(f" 🛡️ [bold yellow][V15.5 守护地球版][/bold yellow] | 并发:{len(cur_pos)}/{MAX_POSITIONS} | 可用:{available/(wallet+1e-9):.1%} | 日盈亏:{daily_pnl:+.2%}")
            tbl = Table(header_style="bold cyan", box=None, padding=(0,1))
            for c in ["战区","板块","方向","军衔","杠杆","占用本金(U)","收益率(%)","浮盈(U)","强平线"]: tbl.add_column(c, justify="center")
            for n, p in cur_pos.items():
                if n not in SYNCED_POSITIONS:
                    SYNCED_POSITIONS.add(n)
                    ENTRY_PRICE_LOG[n] = float(p.get('entryPrice') or 0)
                    POSITION_TIMERS[n] = cur_t
                    ACTUAL_LEVERAGE[n] = 20 if n in MAJORS else 5
                    LEVEL_LOG[n] = 1
                    BASE_SIZE_LOG[n] = ENTRY_PRICE_LOG[n] * abs(float(p.get('contracts',0)))
                    df15_sync = get_data_cached(p['symbol'], '15m')
                    ENTRY_ATR_LOG[n] = df15_sync['atr'].iloc[-1] if df15_sync is not None else (ENTRY_PRICE_LOG[n] * 0.02)
                    DYNAMIC_SL_LOG[n] = ENTRY_PRICE_LOG[n] - ATR_SL_MULTIPLIER * ENTRY_ATR_LOG[n] if p['side'].lower() == 'long' else ENTRY_PRICE_LOG[n] + ATR_SL_MULTIPLIER * ENTRY_ATR_LOG[n]
                    save_state()
                unp = float(p.get('unrealizedPnl', 0)) if 'unrealizedPnl' in p else float(p.get('info', {}).get('unRealizedProfit', 0))
                status = "半仓巡航" if n in PARTIAL_CLOSED else f"全仓 L{LEVEL_LOG.get(n,1)}"
                sector = SECTOR_MAP.get(n, 'ALT')
                sl_disp = f"{DYNAMIC_SL_LOG.get(n, 0):.4f}"
                tbl.add_row(n, sector, "多" if p['side'].lower() == 'long' else "空", status, f"{ACTUAL_LEVERAGE.get(n,1)}x", f"{ENTRY_PRICE_LOG.get(n,0)*abs(float(p.get('contracts',0)))/ACTUAL_LEVERAGE.get(n,1):.2f}", f"{(unp/(wallet/10+1e-9))*100:+.2f}%", f"{unp:+.2f}", f"[bold red]{sl_disp}[/bold red]")
            console.print(tbl)
            LAST_DISPLAY = cur_t

        # 持仓管理
        for n, p in list(cur_pos.items()):
            sym = p['symbol']
            side = p['side'].lower()
            amt = abs(float(p.get('contracts',0)))
            ep = ENTRY_PRICE_LOG.get(n,0)
            mk = float(p.get('markPrice',0))
            pm = (mk-ep)/(ep+1e-9) if side=='long' else (ep-mk)/(ep+1e-9)
            lvl = LEVEL_LOG.get(n,1)
            dur = cur_t - POSITION_TIMERS.get(n, cur_t)

            should_close = False

            if lvl == 1 and (pm <= -0.03 or (dur > DECAY_TIME and pm < 0.003)):
                should_close = True

            if lvl == 3 and abs(pm) < 0.01:
                should_close = True

            if n in DYNAMIC_SL_LOG:
                sl = DYNAMIC_SL_LOG[n]
                if (side == 'long' and mk <= sl) or (side == 'short' and mk >= sl):
                    should_close = True

            df15 = get_data_cached(sym, '15m')
            e144_15m = df15['ema144'].iloc[-1] if df15 is not None else 0
            if (side == 'long' and mk < e144_15m) or (side == 'short' and mk > e144_15m):
                should_close = True

            if should_close:
                if armor_close(sym, side, amt):
                    clear_state(n, is_cooldown=True)
                    save_state()
                continue

            if n not in PARTIAL_CLOSED and pm >= 0.025:
                HIGH_WATER_MARK[n] = max(HIGH_WATER_MARK.get(n, mk), mk) if side=='long' else min(HIGH_WATER_MARK.get(n, mk), mk)
                if abs(HIGH_WATER_MARK[n] - mk) / (HIGH_WATER_MARK[n] + 1e-9) >= 0.012:
                    if armor_close(sym, side, amt * 0.5):
                        PARTIAL_CLOSED[n] = True
                        LEVEL_LOG[n] = 99
                        save_state()
                        continue

            if lvl < 3 and n not in PARTIAL_CLOSED and available/wallet > 0.1 and pm >= 0.02 * lvl:
                fire_side = 'buy' if side == 'long' else 'sell'
                if assassin_fire(sym, fire_side, BASE_SIZE_LOG.get(n, 0), ACTUAL_LEVERAGE.get(n,5)):
                    LEVEL_LOG[n] += 1
                    save_state()

        # 雷达扫描
        if CAN_TRADE_TODAY and len(cur_pos) < MAX_POSITIONS:
            active_sectors = [SECTOR_MAP.get(x, 'ALT') for x in cur_pos.keys()]
            st_t = cur_t
            sc_cnt = 0
            while sc_cnt < len(SYMBOL_LIST):
                sym = SYMBOL_LIST[SCAN_INDEX]
                SCAN_INDEX = (SCAN_INDEX + 1) % len(SYMBOL_LIST)
                sc_cnt += 1
                n = sym.replace('/USDT','')
                if n in cur_pos or n in LEVEL_LOG or (n in COOLDOWN_LOG and cur_t < COOLDOWN_LOG[n]): continue

                target_sector = SECTOR_MAP.get(n, 'ALT')
                if active_sectors.count(target_sector) >= 2 and target_sector != 'ALT': continue

                df4 = get_data_cached(sym, '4h')
                df15 = get_data_cached(sym, '15m')
                if df4 is None or df15 is None: continue

                c15 = df15['c'].iloc[-1]
                vh = max(df4['ema144'].iloc[-1], df4['ema169'].iloc[-1])
                vl = min(df4['ema144'].iloc[-1], df4['ema169'].iloc[-1])
                e576 = df4['ema576'].iloc[-1]

                is_in_tunnel = (vl <= c15 <= vh)
                gap_ok = (vh - vl) / c15 >= TUNNEL_GAP_MIN

                side = None
                if df15['ema14'].iloc[-1] > df15['ema28'].iloc[-1] and c15 > df15['h10'].iloc[-1]:
                    if (not is_in_tunnel) or (is_in_tunnel and gap_ok and c15 > df15['ema20_p'].iloc[-1]):
                        side = 'buy'
                elif df15['ema14'].iloc[-1] < df15['ema28'].iloc[-1] and c15 < df15['l10'].iloc[-1]:
                    if (not is_in_tunnel) or (is_in_tunnel and gap_ok and c15 < df15['ema20_p'].iloc[-1]):
                        side = 'sell'

                if side == 'buy' and c15 < e576: side = None
                if side == 'sell' and c15 > e576: side = None
                if side and abs(get_funding_rate_cached(sym)) > 0.0015: side = None

                if side and df15['v'].iloc[-1] > df15['vol_sma20'].iloc[-2] * 1.3:
                    ema144_15m = df15['ema144'].iloc[-1]
                    bias = abs(c15 - ema144_15m) / (ema144_15m + 1e-9)
                    vol_r = df15['v'].iloc[-1] / df15['vol_sma20'].iloc[-2] if not pd.isna(df15['vol_sma20'].iloc[-2]) and df15['vol_sma20'].iloc[-2] > 0 else 1.0
                    mom = (c15 - df15['c'].iloc[-5]) / df15['c'].iloc[-5]
                    score = float(GLOBAL_AI_MODEL.predict(pd.DataFrame([{'bias': bias, 'vol_r': vol_r, 'mom': mom}]))[0])
                    ls_ratio = get_ls_ratio(sym)
                    if (side == 'buy' and ls_ratio > 1.8) or (side == 'sell' and ls_ratio < 0.6):
                        score -= 15
                    if score >= AI_THRESHOLD:
                        lev = 20 if n in MAJORS else 5
                        curr_atr = df15['atr'].iloc[-1]
                        sz = min((wallet * RISK_PCT) / (curr_atr / c15 if curr_atr > 0 else 0.01),
                                 (wallet * MAX_MARGIN_PER_TRADE) * lev)
                        if assassin_fire(sym, side, sz, lev):
                            LEVEL_LOG[n] = 1
                            BASE_SIZE_LOG[n] = sz
                            POSITION_TIMERS[n] = time.time()
                            ENTRY_PRICE_LOG[n] = c15
                            ENTRY_ATR_LOG[n] = curr_atr
                            DYNAMIC_SL_LOG[n] = c15 - ATR_SL_MULTIPLIER * curr_atr if side == 'buy' else c15 + ATR_SL_MULTIPLIER * curr_atr
                            save_state()
                            active_sectors.append(target_sector)
                            break
                if time.time() - st_t > 2.0: break
    except Exception as e:
        print(f"🚨 [CRITICAL] 引擎异常: {e}\n{traceback.format_exc()}")
