def _send_alert_sync(msg):

    except: pass

def send_alert(msg):
    threading.Thread(target=_send_alert_sync, args=(msg,), daemon=True).start()

def update_funding_rates():

        except: pass

def get_data_cached(sym, tf):
    now = time.time()
    cache_key = f"{sym}_{tf}"
    if cache_key in DATA_CACHE and (now - DATA_CACHE[cache_key]['time']) < (120 if tf == '15m' else 600): return DATA_CACHE[cache_key]['data']
    try:
        ohlcv = exchange.fetch_ohlcv(sym, tf, limit=1000 if tf == '4h' else 500)
        df = pd.DataFrame(ohlcv, columns=['t','o','h','l','c','v'])
        if tf == '4h': df['ema576'] = df['c'].ewm(span=576).mean()
        if tf == '15m':
            df['ema14'], df['ema21'], df['ema60'] = df['c'].ewm(span=14).mean(), df['c'].ewm(span=21).mean(), df['c'].ewm(span=60).mean()
            df['vol_sma20'] = df['v'].rolling(20).mean()
            df['prev_c'] = df['c'].shift(1)
            df['tr'] = df[['h','l','prev_c']].apply(lambda x: max(x['h']-x['l'], abs(x['h']-x['prev_c']), abs(x['l']-x['prev_c'])) if not pd.isna(x['prev_c']) else x['h']-x['l'], axis=1)
            df['atr'] = df['tr'].ewm(alpha=1/14).mean()
        DATA_CACHE[cache_key] = {'time': now, 'data': df}
        return df
    except: return None

def clear_state(n, is_cooldown=True):
    for d in [LEVEL_LOG, BASE_SIZE_LOG, HIGH_WATER_MARK, POSITION_TIMERS, ENTRY_PRICE_LOG, TARGET_PRICE_LOG, ACTUAL_LEVERAGE, DYNAMIC_SL_LOG, PARTIAL_CLOSED, ENTRY_ATR_LOG]: d.pop(n, None)
    SYNCED_POSITIONS.discard(n)
    if is_cooldown: COOLDOWN_LOG[n] = time.time() + COOLDOWN_TIME

    for file_name in fallback_files:
        if os.path.exists(file_name):
            try:

def assassin_fire(sym, side, nom_u, lev):
    try:
        logger.info(f"👉 发起进攻: {sym} | 方向: {side} | 拟投兵力: {nom_u:.2f}U")
        try: exchange.set_leverage(lev, sym)
        except: pass
        p = exchange.fetch_ticker(sym).get('last')
        
        m = exchange.market(sym)
        max_mkt_qty = None
        for f in m['info']['filters']:
            if f['filterType'] == 'MARKET_LOT_SIZE':
                max_mkt_qty = float(f['maxQty'])
                break
        
        if not max_mkt_qty: max_mkt_qty = m['limits']['amount']['max']
        
        target_qty = nom_u / p
        if max_mkt_qty and target_qty > max_mkt_qty:
            logger.warning(f"⚠️ {sym} 拟投量超过上限，强制降压。")
            target_qty = max_mkt_qty * 0.98 
            
        amt = float(exchange.amount_to_precision(sym, target_qty))
        exchange.create_market_order(sym, 'buy' if side=='buy' else 'sell', amt)
        return True
    except Exception as e:
        logger.error(f"❌ 进攻受阻 ({sym}): {e}")
        n = sym.split('/')[0]
        COOLDOWN_LOG[n] = time.time() + 1800
        return False

def armor_close(sym, side, amt):
    try:
        logger.info(f"🛡️ 启动撤退: {sym} | 数量: {amt}")
        exchange.create_market_order(sym, 'sell' if side == 'long' else 'buy', amt, params={'reduceOnly': True})
        return True
    except: return False
