import os
from dotenv import load_dotenv
from crewai import Agent, Task, Crew, Process
from crewai_tools import SerperDevTool

# 加载 .env 里的最高机密 (API Keys)
load_dotenv()

# ==========================================
# ⚔️ 硅基远征军 - Alpha先遣队配置
# ==========================================

# 装备库：给军情局配备的全球搜索雷达
search_tool = SerperDevTool()

print("🚀 统帅部指令下达：硅基远征军 Alpha 小队，正在唤醒...")

# ------------------------------------------
# 👁️ L1 层级：全域军情局特工 (只负责找数据，绝对客观)
# ------------------------------------------
intel_agent = Agent(
    role='首席情报特工 (军情局)',
    goal='精确抓取目标资产的实时价格、核心基本面新闻及市场情绪。',
    backstory=(
        "你是硅基远征军军情局的王牌特工。你像一台没有感情的机器，"
        "绝不提供任何主观交易建议，只负责用雷达(Serper)把最真实的战场数据带回来。"
    ),
    verbose=True,
    allow_delegation=False,
    tools=[search_tool] # 只有军情局配发了搜索工具，其他部门不准上网
)

# ------------------------------------------
# 🗡️ L2 层级：金融重装军团宽客 (干活的野战军)
# ------------------------------------------
quant_agent = Agent(
    role='高频量化宽客 (金融军团)',
    goal='根据军情局的情报，计算具体的开平仓点位，并生成实盘交易策略。',
    backstory=(
        "你是前华尔街顶级交易员，现隶属硅基金融军团。"
        "你极度渴望开仓赚钱。你需要根据情报，写出一份包含【入场价、止损价、盈亏比】的具体作战方案。"
        "注意：你的方案必须明确写出止损的回撤百分比（如止损幅度为1.5%）。"
    ),
    verbose=True,
    allow_delegation=False
)

# ------------------------------------------
# ⚖️ L1 层级：最高宪兵督察 (一票否决权)
# ------------------------------------------
mp_agent = Agent(
    role='风控与合规宪兵局长 (督战委员会)',
    goal='审查宽客提交的策略。坚决死守 2% 回撤红线及周末禁令，违规直接打回。',
    backstory=(
        "你是远征军中最冷酷的执刑者。你直接向总司令汇报。"
        "你的任务是拿着放大镜看宽客的报告："
        "1. 如果报告建议的止损幅度超过总资金的 2%，直接否决该策略，并在最终报告中写明【风控驳回】。"
        "2. 确保没有盲目追高的情绪化建议。"
        "如果合规，你将盖上【风控通过】的印章；如果不合规，你将拦截该交易。"
    ),
    verbose=True,
    allow_delegation=False
)

# ==========================================
# 📋 作战任务书 (SOP 流转逻辑)
# ==========================================

# 目标标的：您可以随时修改这里，比如换成 "BTC" 或 "特斯拉股票"
TARGET_ASSET = "黄金 (XAU/USD)" 

task1_intel = Task(
    description=f'使用搜索工具，查明 {TARGET_ASSET} 当前的最新价格，以及过去24小时内影响它的两大核心宏观新闻。',
    expected_output=f'一份关于 {TARGET_ASSET} 的标准化军情报告，包含现价和核心新闻摘要。',
    agent=intel_agent
)

task2_analysis = Task(
    description=(
        f'基于军情局的报告，制定 {TARGET_ASSET} 的交易计划。'
        f'必须包含：1. 趋势判断；2. 具体的入场位；3. 具体的止损位；4. 明确计算出止损所占的百分比。'
    ),
    expected_output='一份带具体点位和风控数据的量化交易策略报告。',
    agent=quant_agent
)

task3_audit = Task(
    description=(
        '审查金融军团提交的交易策略。严格计算其止损百分比是否 > 2%。'
        '如果超过 2%，你的最终报告必须是警告信，驳回交易；'
        '如果小于 2% 且逻辑合理，输出最终的盖章版《远征军执行报告》。'
    ),
    expected_output='最终呈报给总司令的《审查后作战报告》（明确标识【风控通过】或【风控驳回】）。',
    agent=mp_agent
)

# ==========================================
# 🔗 组建集团军群并合闸执行
# ==========================================
alpha_crew = Crew(

    verbose=True
)

# 总司令点火
print(f"🔥 目标锁定：{TARGET_ASSET}。各单位就位，战役开始！\n")
result = alpha_crew.kickoff()

print("==========================================")
print("👑 呈报总司令的最终战果：")

import os
import subprocess  # 用于强行弹出文件
from dotenv import load_dotenv
from crewai import Agent, Task, Crew, Process
from crewai_tools import SerperDevTool

# 加载机密配置
load_dotenv()

# ==========================================
# ⚔️ 硅基远征军 - 自动化指挥中心升级版
# ==========================================

search_tool = SerperDevTool()

print("🚀 统帅部指令下达：硅基远征军 Alpha 小队正在集结...")

# 1. 军情局
intel_agent = Agent(
    role='首席情报特工',
    goal='抓取 {target} 的实时数据。',
    backstory="你是远征军的眼睛，只负责真实数据的采集。",
    verbose=True,
    tools=[search_tool]
)

# 2. 金融军团 (宽客)
quant_agent = Agent(
    role='量化狙击手',
    goal='根据情报制定具体的入场和止损策略。',
    backstory="你是远征军的利剑，负责在数据中寻找杀机。止损比例必须明确计算。",
    verbose=True
)

# 3. 最高宪兵委 (督战)
mp_agent = Agent(
    role='最高宪兵局长',
    goal='审计策略安全性，死守 2% 风险红线。',
    backstory="你是远征军的刹车，拥有最高否决权。你的报告是最终执行依据。",
    verbose=True
)

# 定义作战任务
TARGET_ASSET = "黄金 (Gold XAU/USD)"

task1 = Task(
    description=f"搜集 {TARGET_ASSET} 的最新价格和影响因素。",
    expected_output="实时军情数据摘要。",
    agent=intel_agent
)

task2 = Task(
    description=f"制定 {TARGET_ASSET} 交易计划，明确止损比例。",
    expected_output="详细的量化作战方案。",
    agent=quant_agent
)

task3 = Task(
    description="最终审计：止损 > 2% 则驳回，否则批准执行。",
    expected_output="总司令亲启的最终审计战报。",
    agent=mp_agent
)

# 组建军团
alpha_crew = Crew(
    agents=[intel_agent, quant_agent, mp_agent],
    tasks=[task1, task2, task3],
    process=Process.sequential,
    verbose=True
)

# 执行作战
print(f"🔥 战役开始：目标 {TARGET_ASSET}")
final_result = alpha_crew.kickoff()

# ==========================================
# 📬 传令兵系统：自动生成并【强制弹出】战报
# ==========================================
report_name = "DEPT_REPORT_LATEST.md"

# 1. 整理战报内容
report_content = f"""# 📜 硅基远征军：核心作战报告
---
**目标标的:** {TARGET_ASSET}
**执行日期:** 2026年4月26日

## 📑 统帅部审计结论
{final_result}

---
*注：本报告由最高宪兵委自动签发，抄送总司令。*
"""

# 2. 写入本地文件
with open(report_name, "w", encoding="utf-8") as f:
    f.write(report_content)

print(f"\n✅ 战报已送达！正在强制弹出...")

# 3. 核心绝招：根据不同系统强行弹出文件
try:
    if os.name == 'nt':  # Windows
        os.startfile(report_name)
    elif os.uname().sysname == 'Darwin':  # MacOS
        subprocess.call(('open', report_name))
    else:  # Linux
        subprocess.call(('xdg-open', report_name))
except Exception as e:
    print(f"❌ 自动弹出失败，请手动查看 {report_name}。报错：{e}")
